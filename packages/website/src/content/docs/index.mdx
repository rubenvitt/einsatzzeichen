import { loadSnapshot } from '../../lib/snapshot';

export const snapshot = loadSnapshot();
export const matrix = snapshot.coverage.matrix;
export const technicallyReviewed = matrix.filter((row) => row.technical.status === 'approved').length;
export const withDeviation = matrix.filter((row) => row.technical.status === 'deviation').length;
export const domainApproved = matrix.filter((row) => row.domain.status === 'approved').length;
export const openDomainAreas = snapshot.coverage.openDomainReviewsByArea.length;
export const openDomainTotal = snapshot.coverage.openDomainReviewsByArea.reduce(
  (sum, area) => sum + area.count,
  0,
);

Die beiden Zahlen zählen Verschiedenes, deshalb stehen sie getrennt: eine **Manifestzeile** ist ein
Abschnitt der Baseline mit seiner Umsetzung — davon gibt es {matrix.length}. Eine **Zeichenseite**
gibt es nur für die {snapshot.symbols.length} Zeilen, die ein eigenes Zeichen ergeben; Elemente und
Piktogramme haben keine, weil sie allein kein Zeichen sind. Wer eine Vollständigkeitsaussage liest,
sollte wissen, worauf sie sich bezieht.
