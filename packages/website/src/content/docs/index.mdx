import { loadSnapshot } from '../../lib/snapshot';
import { reviewSummaryOfRows } from '../../lib/review-summary';
import { formatReviewDate } from '../../components/StatusPair.tsx';

export const snapshot = loadSnapshot();
export const matrix = snapshot.coverage.matrix;
// Zwei Zählmengen, bewusst getrennt (`review-summary.ts`): die Prüfliste zählt jeden geprüften
// Abschnitt der Vorlage, also auch Bausteine ohne eigene Seite; `symbols` zählt die Zeichen mit
// eigener Seite. Wer beides vermischt, nennt eine Zahl, die es nicht gibt.
export const listed = reviewSummaryOfRows(matrix);
export const clarifiedSources = snapshot.sources.filter((source) => source.status === 'clarified').length;
// Das Datum der fachlichen Prüfung, sofern sie an allen Einträgen dasselbe trägt. Kein fester
// Text: steht später an einem Eintrag ein anderes Datum, fällt die Angabe weg statt falsch zu sein.
export const domainDates = [...new Set(matrix.map((row) => row.domain.date).filter((date) => date !== undefined))];
export const domainReviewers = [...new Set(matrix.map((row) => row.domain.reviewer).filter((name) => name !== undefined))];

## Was hier steht

Dieses Projekt baut die taktischen Zeichen des Bevölkerungsschutzes aus ihren Bestandteilen —
Grundzeichen, Organisation, Stärke, Fähigkeiten — statt sie als fertige Bilddateien abzulegen.
Für dich heißt das: du kannst jedes Zeichen ansehen, herunterladen und weiterverwenden, und du
kannst dir eines zusammensetzen, das es als fertige Datei nirgends gibt.

Zu jedem Zeichen steht seine Fundstelle in der Vorlage und sein Prüfstand. Der Prüfstand hat zwei
Seiten, und sie sagen Verschiedenes: **technisch geprüft** heißt, das Bild stimmt mit der Vorlage
überein. **Fachlich geprüft** heißt, jemand hat bestätigt, dass es das richtige Zeichen für diese
Bedeutung ist.

Die ersten drei Zahlen zählen Abschnitte der Vorlage, nicht Zeichen: ein Abschnitt kann ein
Zeichen ergeben oder einen Baustein, der erst in einem Zeichen sichtbar wird. Eine eigene Seite
haben nur die {snapshot.symbols.length} Zeichen. Wer eine Vollständigkeitsaussage liest, sollte
wissen, worauf sie sich bezieht — deshalb stehen beide Zahlen hier und nicht eine gemittelte.
