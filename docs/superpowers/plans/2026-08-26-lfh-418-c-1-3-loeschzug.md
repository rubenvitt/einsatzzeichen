  import { mkdirSync, readFileSync, writeFileSync } from 'node:fs';
  import { dirname, resolve } from 'node:path';

  const referencePath = process.argv[2];
  if (referencePath === undefined) {

  }

  const outputs = [
    {
      source: resolve('packages/catalog/src/__snapshots__/multi-size/recipe.C.1.3.svg'),
      target: resolve('docs/reviews/assets/2026-08-26-c-1-3-catalog.png'),

    {
      source: resolve(referencePath),
      target: resolve('out/lfh-418/reference.png'),
      width: 900,
    },
  ];

  for (const output of outputs) {
    mkdirSync(dirname(output.target), { recursive: true });
    const svg = readFileSync(output.source);
    const png = new Resvg(svg, {
      fitTo: { mode: 'width', value: output.width },
      font: { loadSystemFonts: true },
    }).render().asPng();
    writeFileSync(output.target, png);
  }

  const hashLines = outputs.map(({ target }) => {
    const digest = createHash('sha256').update(readFileSync(target)).digest('hex');
    return `${digest}  ${target}`;
  });
  writeFileSync(resolve('out/lfh-418/sha256.txt'), `${hashLines.join('\n')}\n`);
  console.log(hashLines.join('\n'));
